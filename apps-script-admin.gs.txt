var PRODUCTS_SHEET_NAME = "Sayfa1";
var SETTINGS_SHEET_NAME = "Ayarlar";
var ORDERS_SPREADSHEET_ID = "11DQF5rjHUNcRi9_uAIySmmjXmzrB1Kd_w42irh2h3Gc";
var ORDERS_SHEET_NAME = "Siparişler";
var ADMIN_PASSWORD_PROPERTY = "ADMIN_PASSWORD"; // Proje Ayarları > Script Properties'te tanımlanır

function checkPassword(pw) {
  var stored = PropertiesService.getScriptProperties().getProperty(ADMIN_PASSWORD_PROPERTY);
  return stored && pw === stored;
}

function doGet(e) {
  var action = (e.parameter && e.parameter.action) || "listProducts";
  if (action === "listProducts") return jsonResponse({ result: "ok", data: getAllProducts() });
  if (action === "listSettings") return jsonResponse({ result: "ok", data: getAllSettings() });
  if (action === "listOrders") {
    if (!checkPassword(e.parameter.password)) return jsonResponse({ result: "error", message: "Şifre yanlış." });
    return jsonResponse({ result: "ok", data: getAllOrders() });
  }
  return jsonResponse({ result: "error", message: "Bilinmeyen işlem: " + action });
}

function doPost(e) {
  var body;
  try { body = JSON.parse(e.postData.contents); }
  catch (err) { return jsonResponse({ result: "error", message: "Geçersiz istek." }); }

  if (!checkPassword(body.password)) {
    return jsonResponse({ result: "error", message: "Şifre yanlış." });
  }

  switch (body.action) {
    case "upsertProduct": return jsonResponse(upsertProduct(body.product));
    case "deleteProduct": return jsonResponse(deleteProduct(body.id));
    case "updateSettings": return jsonResponse(updateSettings(body.settings));
    case "updateOrderStatus": return jsonResponse(updateOrderStatus(body.orderId, body.status));
    default: return jsonResponse({ result: "error", message: "Bilinmeyen işlem: " + body.action });
  }
}

// ---------- Sipariş işlemleri ----------

function getOrdersSheet() {
  var ss = SpreadsheetApp.openById(ORDERS_SPREADSHEET_ID);
  return ss.getSheetByName(ORDERS_SHEET_NAME);
}

function getAllOrders() {
  var sheet = getOrdersSheet();
  if (!sheet) return [];
  var data = sheet.getDataRange().getValues();
  var headers = data[0];
  var orders = data.slice(1)
    .filter(function (r) { return r[1]; }) // Sipariş No dolu olanlar
    .map(function (r) {
      var o = {};
      headers.forEach(function (h, i) { o[h] = r[i]; });
      return o;
    });
  orders.reverse(); // en yeni sipariş en üstte
  return orders;
}

function updateOrderStatus(orderId, status) {
  if (!orderId || !status) return { result: "error", message: "Sipariş no ve durum zorunlu." };
  var sheet = getOrdersSheet();
  if (!sheet) return { result: "error", message: "Siparişler sekmesi bulunamadı." };
  var data = sheet.getDataRange().getValues();
  var headers = data[0];
  var idCol = headers.indexOf("Sipariş No");
  var statusCol = headers.indexOf("Durum");
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][idCol]) === String(orderId)) {
      sheet.getRange(i + 1, statusCol + 1).setValue(status);
      return { result: "ok" };
    }
  }
  return { result: "error", message: "Sipariş bulunamadı." };
}

// ---------- Ürün işlemleri ----------

function getProductsSheet() {
  return SpreadsheetApp.getActiveSpreadsheet().getSheetByName(PRODUCTS_SHEET_NAME);
}

function getAllProducts() {
  var sheet = getProductsSheet();
  var data = sheet.getDataRange().getValues();
  var headers = data[0];
  return data.slice(1)
    .filter(function (r) { return r[0]; })
    .map(function (r) {
      var o = {};
      headers.forEach(function (h, i) { o[h] = r[i]; });
      return o;
    });
}

function upsertProduct(product) {
  if (!product || !product.id) return { result: "error", message: "Ürün ID zorunlu." };
  var sheet = getProductsSheet();
  var data = sheet.getDataRange().getValues();
  var headers = data[0];
  var idCol = headers.indexOf("id");
  var rowValues = headers.map(function (h) {
    return product[h] !== undefined ? product[h] : "";
  });
  var foundRow = -1;
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][idCol]) === String(product.id)) { foundRow = i + 1; break; }
  }
  if (foundRow > -1) sheet.getRange(foundRow, 1, 1, rowValues.length).setValues([rowValues]);
  else sheet.appendRow(rowValues);
  return { result: "ok" };
}

function deleteProduct(id) {
  if (!id) return { result: "error", message: "Ürün ID zorunlu." };
  var sheet = getProductsSheet();
  var data = sheet.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][0]) === String(id)) { sheet.deleteRow(i + 1); break; }
  }
  return { result: "ok" };
}

// ---------- Ayarlar işlemleri ----------

function getSettingsSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(SETTINGS_SHEET_NAME);
  if (!sheet) return null;
  return sheet;
}

function getAllSettings() {
  var sheet = getSettingsSheet();
  if (!sheet) return {};
  var data = sheet.getDataRange().getValues();
  var out = {};
  for (var i = 1; i < data.length; i++) {
    if (data[i][0]) out[data[i][0]] = data[i][1];
  }
  return out;
}

function updateSettings(settings) {
  if (!settings) return { result: "error", message: "Ayar verisi eksik." };
  var sheet = getSettingsSheet();
  if (!sheet) return { result: "error", message: "'Ayarlar' sekmesi bulunamadı." };
  var data = sheet.getDataRange().getValues();
  var keyToRow = {};
  for (var i = 1; i < data.length; i++) { if (data[i][0]) keyToRow[data[i][0]] = i + 1; }

  Object.keys(settings).forEach(function (key) {
    var value = settings[key];
    if (keyToRow[key]) {
      sheet.getRange(keyToRow[key], 2).setValue(value);
    } else {
      sheet.appendRow([key, value]);
    }
  });
  return { result: "ok" };
}

// ---------- Yardımcı ----------

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
