var ORDERS_SHEET_NAME = "Siparişler";
var ADMIN_PASSWORD_PROPERTY = "ADMIN_PASSWORD"; // Proje Ayarları > Script Properties'te tanımlanır

function checkPassword(pw) {
  var stored = PropertiesService.getScriptProperties().getProperty(ADMIN_PASSWORD_PROPERTY);
  return stored && pw === stored;
}

function doGet(e) {
  var action = (e.parameter && e.parameter.action) || "listProducts";
  if (action === "listProducts") return jsonResponse({ result: "ok", data: getAllProducts() });
  if (action === "listSettings") return jsonResponse({ result: "ok", data: getAllSettings() });
  if (action === "listCampaigns") {
    if (!checkPassword(e.parameter.password)) return jsonResponse({ result: "error", message: "Şifre yanlış." });
    return jsonResponse({ result: "ok", data: getAllCampaigns() });
  }
  if (action === "listOrders") {
    if (!checkPassword(e.parameter.password)) return jsonResponse({ result: "error", message: "Şifre yanlış." });
    return jsonResponse({ result: "ok", data: getAllOrders() });
  }
  return jsonResponse({ result: "error", message: "Bilinmeyen işlem: " + action });
}

function doPost(e) {
  var body;
  try { body = JSON.parse(e.postData.contents); }
  catch (err) { return jsonResponse({ result: "error", message: "Geçersiz istek." }); }

  if (!checkPassword(body.password)) {
    return jsonResponse({ result: "error", message: "Şifre yanlış." });
  }

  switch (body.action) {
    case "upsertProduct": return jsonResponse(withFeedRefresh_(upsertProduct(body.product)));
    case "deleteProduct": return jsonResponse(withFeedRefresh_(deleteProduct(body.id)));
    case "rebuildFeeds": return jsonResponse(rebuildFeeds());
    case "upsertCampaign": return jsonResponse(withFeedRefresh_(upsertCampaign(body.campaign)));
    case "deleteCampaign": return jsonResponse(withFeedRefresh_(deleteCampaign(body.id)));
    case "updateSettings": return jsonResponse(updateSettings(body.settings));
    case "updateOrderStatus": return jsonResponse(updateOrderStatus(body.orderId, body.status));
    case "listBlog": return jsonResponse({ result: "ok", data: getAllBlog_() });
    case "upsertBlog": return jsonResponse(upsertBlog_(body.post));
    case "deleteBlog": return jsonResponse(deleteBlog_(body.slug));
    case "metaStatus":
    case "metaSaveConfig":
    case "metaDisconnect":
    case "metaDashboard":
    case "metaSetStatus":
    case "metaSetBudget":
    case "metaAdSetup":
    case "metaSaveAdSetup":
    case "metaIgMedia":
    case "metaCreateAd":
    case "metaCatalogs":
      return jsonResponse(handleMetaAction_(body));
    default: return jsonResponse({ result: "error", message: "Bilinmeyen işlem: " + body.action });
  }
}

// ---------- Sipariş işlemleri ----------

function getOrdersSheet() {
  var ss = SpreadsheetApp.openById(ORDERS_SPREADSHEET_ID);
  return ss.getSheetByName(ORDERS_SHEET_NAME);
}

function getAllOrders() {
  var sheet = getOrdersSheet();
  if (!sheet) return [];
  var data = sheet.getDataRange().getValues();
  var headers = data[0];
  var orders = data.slice(1)
    .filter(function (r) { return r[1]; }) // Sipariş No dolu olanlar
    .map(function (r) {
      var o = {};
      headers.forEach(function (h, i) { o[h] = r[i]; });
      return o;
    });
  orders.reverse(); // en yeni sipariş en üstte
  return orders;
}

function updateOrderStatus(orderId, status) {
  if (!orderId || !status) return { result: "error", message: "Sipariş no ve durum zorunlu." };
  var sheet = getOrdersSheet();
  if (!sheet) return { result: "error", message: "Siparişler sekmesi bulunamadı." };
  var data = sheet.getDataRange().getValues();
  var headers = data[0];
  var idCol = headers.indexOf("Sipariş No");
  var statusCol = headers.indexOf("Durum");
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][idCol]) === String(orderId)) {
      sheet.getRange(i + 1, statusCol + 1).setValue(status);
      return { result: "ok" };
    }
  }
  return { result: "error", message: "Sipariş bulunamadı." };
}

// ---------- Ürün işlemleri ----------

function getProductsSheet() {
  return SpreadsheetApp.getActiveSpreadsheet().getSheetByName(PRODUCTS_SHEET_NAME);
}

function getAllProducts() {
  var sheet = getProductsSheet();
  var data = sheet.getDataRange().getValues();
  var headers = data[0];
  return data.slice(1)
    .filter(function (r) { return r[0]; })
    .map(function (r) {
      var o = {};
      headers.forEach(function (h, i) { o[h] = r[i]; });
      return o;
    });
}

function upsertProduct(product) {
  if (!product || !product.id) return { result: "error", message: "Ürün ID zorunlu." };
  var sheet = getProductsSheet();
  var data = sheet.getDataRange().getValues();
  var headers = data[0];
  var idCol = headers.indexOf("id");
  var foundRow = -1;
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][idCol]) === String(product.id)) { foundRow = i + 1; break; }
  }
  // Gönderilmeyen sütunlar mevcut değerini korur (silinmez)
  var rowValues = headers.map(function (h, c) {
    if (product[h] !== undefined) return product[h];
    return foundRow > -1 ? data[foundRow - 1][c] : "";
  });
  if (foundRow > -1) sheet.getRange(foundRow, 1, 1, rowValues.length).setValues([rowValues]);
  else sheet.appendRow(rowValues);
  return { result: "ok" };
}

function deleteProduct(id) {
  if (!id) return { result: "error", message: "Ürün ID zorunlu." };
  var sheet = getProductsSheet();
  var data = sheet.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][0]) === String(id)) { sheet.deleteRow(i + 1); break; }
  }
  return { result: "ok" };
}

// ---------- Ayarlar işlemleri ----------

function getSettingsSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(SETTINGS_SHEET_NAME);
  if (!sheet) return null;
  return sheet;
}

function getAllSettings() {
  var sheet = getSettingsSheet();
  if (!sheet) return {};
  var data = sheet.getDataRange().getValues();
  var out = {};
  for (var i = 1; i < data.length; i++) {
    if (data[i][0]) out[data[i][0]] = data[i][1];
  }
  return out;
}

function updateSettings(settings) {
  if (!settings) return { result: "error", message: "Ayar verisi eksik." };
  var sheet = getSettingsSheet();
  if (!sheet) return { result: "error", message: "'Ayarlar' sekmesi bulunamadı." };
  var data = sheet.getDataRange().getValues();
  var keyToRow = {};
  for (var i = 1; i < data.length; i++) { if (data[i][0]) keyToRow[data[i][0]] = i + 1; }

  Object.keys(settings).forEach(function (key) {
    var value = settings[key];
    if (keyToRow[key]) {
      sheet.getRange(keyToRow[key], 2).setValue(value);
    } else {
      sheet.appendRow([key, value]);
    }
  });
  return { result: "ok" };
}

var SITE_URL = "https://www.gulsumaydin.com.tr";
var FEED_BRAND = "Gülsüm Aydın Butik";
var META_FEED_SHEET = "MetaFeed";
var GOOGLE_FEED_SHEET = "GoogleFeed";
var GOOGLE_CATEGORY = "Apparel & Accessories > Clothing";
var GOOGLE_COLUMNS = ["id", "title", "description", "link", "image_link", "availability", "price", "brand",
  "condition", "google_product_category", "sale_price", "color", "size", "gender", "age_group"];
var META_COLUMNS = ["id", "title", "description", "availability", "condition", "price", "link", "image_link",
  "brand", "sale_price", "color", "size", "gender", "age_group", "google_product_category"];

function withFeedRefresh_(result) {
  if (result && result.result === "ok") {
    try { rebuildFeeds(); }
    catch (err) { console.error("Katalog beslemesi güncellenemedi: " + err); }
  }
  return result;
}

// Bir kez elle çalıştırın: beslemeleri saatte bir otomatik yeniler
function setupFeedTrigger() {
  ScriptApp.getProjectTriggers().forEach(function (t) {
    if (t.getHandlerFunction() === "rebuildFeeds") ScriptApp.deleteTrigger(t);
  });
  ScriptApp.newTrigger("rebuildFeeds").timeBased().everyHours(1).create();
}

function rebuildFeeds() {
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(15000)) return { result: "error", message: "Başka bir güncelleme sürüyor, biraz sonra tekrar deneyin." };
  try {
    var camps = getAllCampaigns();
    var items = getAllProducts().map(function (p) { return feedItem_(p, camps); }).filter(function (x) { return x; });
    // Okuma hatası/boş tablo durumunda mevcut beslemeleri silmemek için
    if (!items.length) return { result: "error", message: "Beslemeye uygun ürün bulunamadı; mevcut beslemeler korundu." };
    var g = writeFeed_(GOOGLE_FEED_SHEET, GOOGLE_COLUMNS, items);
    var m = writeFeed_(META_FEED_SHEET, META_COLUMNS, items);
    return { result: "ok", count: items.length, changed: g || m };
  } finally {
    lock.releaseLock();
  }
}

// Fotoğrafı, geçerli fiyatı ve kimliği olan ürünleri besleme satırına çevirir
function feedItem_(p, camps) {
  var id = String(p.id == null ? "" : p.id).trim();
  if (!id) return null;

  var colors = splitList_(p.colors);
  var color = colors.length ? colors[0] : "Standart";
  var photoMap = parseColorImages_(p.colorImages);
  var notVideo = function (u) { return !isVideoUrl_(u); };
  var imgs = (photoMap[color] && photoMap[color].length ? photoMap[color] : splitList_(p.images)).filter(notVideo);
  if (!imgs.length) imgs = splitList_(p.images).filter(notVideo);
  if (!imgs.length) return null; // sadece videosu olan / fotoğrafsız ürünler beslemeye girmez

  var price = parsePrice_(p.price);
  var old = p.oldPrice ? parsePrice_(p.oldPrice) : 0;
  if (!(price > 0)) return null;
  // Aktif fiyat kampanyası varsa sitedeki fiyatla birebir aynı olsun (Merchant fiyat uyuşmazlığı olmasın)
  var hit = priceCampaign_({ id: id, category: p.category, price: price }, camps || []);
  if (hit) { old = price; price = hit.price; }
  var onSale = old > price;

  var sizes = splitList_(p.sizes);
  var desc = String(hasText_(p.description) ? p.description : (hasText_(p.shortDescription) ? p.shortDescription : p.name || ""))
    .replace(/\s+/g, " ").trim().slice(0, 5000);

  return {
    id: id,
    title: String(p.name || id).replace(/\s+/g, " ").trim().slice(0, 150),
    description: desc,
    link: SITE_URL + "/?urun=" + encodeURIComponent(id),
    image_link: absUrl_(imgs[0]),
    availability: "in stock",
    price: (onSale ? old : price).toFixed(2) + " TRY",
    sale_price: onSale ? price.toFixed(2) + " TRY" : "",
    brand: FEED_BRAND,
    condition: "new",
    google_product_category: GOOGLE_CATEGORY,
    color: color,
    size: sizes.length ? sizes.join("/") : "Standart",
    gender: "female",
    age_group: "adult"
  };
}

// Değerler aynıysa yazmaz (gereksiz sürüm geçmişi oluşmasın). Değiştiyse true döner.
function writeFeed_(name, columns, items) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(name) || ss.insertSheet(name);
  var values = [columns].concat(items.map(function (it) {
    return columns.map(function (c) { return it[c] == null ? "" : String(it[c]); });
  }));
  var oldRows = sheet.getLastRow(), oldCols = sheet.getLastColumn();

  if (oldRows === values.length && oldCols === columns.length) {
    var current = sheet.getRange(1, 1, oldRows, oldCols).getValues().map(function (r) {
      return r.map(function (v) { return String(v == null ? "" : v); });
    });
    if (JSON.stringify(current) === JSON.stringify(values)) return false;
  }

  var range = sheet.getRange(1, 1, values.length, columns.length);
  range.setNumberFormat("@"); // "38/40" gibi değerler tarihe çevrilmesin
  range.setValues(values);
  if (oldRows > values.length) {
    sheet.getRange(values.length + 1, 1, oldRows - values.length, Math.max(oldCols, columns.length)).clearContent();
  }
  if (oldCols > columns.length) {
    sheet.getRange(1, columns.length + 1, Math.max(oldRows, values.length), oldCols - columns.length).clearContent();
  }
  return true;
}

function hasText_(v) { v = String(v == null ? "" : v).trim(); return v !== "" && v !== "—"; }

function splitList_(s) {
  return s ? String(s).split("|").map(function (x) { return x.trim(); }).filter(Boolean) : [];
}

function isVideoUrl_(u) { return /\.(mp4|webm|mov)(\?|$)/i.test(u || ""); }

function absUrl_(u) {
  u = String(u || "").trim();
  if (!u) return "";
  return /^https?:\/\//i.test(u) ? u : SITE_URL + "/" + u.replace(/^\/+/, "");
}

// "Siyah=url1|url2;Beyaz=url3" -> {Siyah:[url1,url2], Beyaz:[url3]}
function parseColorImages_(str) {
  var out = {};
  if (!str) return out;
  String(str).split(";").forEach(function (part) {
    var i = part.indexOf("=");
    if (i < 1) return;
    var color = part.slice(0, i).trim();
    var urls = part.slice(i + 1).split("|").map(function (u) { return u.trim(); }).filter(Boolean).slice(0, 5);
    if (color && urls.length) out[color] = urls;
  });
  return out;
}

// Sitedeki parsePrice ile aynı kural (fiyatlar sitede ve beslemede birebir aynı çıksın)
function parsePrice_(str) {
  if (str == null) return 0;
  var s = String(str).trim().replace(/[^0-9.,\-]/g, "");
  if (!s) return 0;
  if (s.indexOf(",") > -1 && s.indexOf(".") > -1) {
    if (s.lastIndexOf(",") > s.lastIndexOf(".")) { s = s.replace(/\./g, "").replace(",", "."); }
    else { s = s.replace(/,/g, ""); }
  } else if (s.indexOf(",") > -1) {
    var parts = s.split(",");
    if (parts[parts.length - 1].length <= 2) { s = parts.slice(0, -1).join("") + "." + parts[parts.length - 1]; }
    else { s = s.replace(/,/g, ""); }
  }
  var n = parseFloat(s);
  return isNaN(n) ? 0 : n;
}

var CAMPAIGNS_SHEET_NAME = "Kampanyalar";
var CAMPAIGN_COLUMNS = ["id", "ad", "tip", "kapsam", "hedef", "adet", "deger", "baslangic", "bitis", "aktif", "aciklama"];
var CAMPAIGN_TYPES = ["yuzde", "tl", "coklu_sabit", "coklu_yuzde"];
var CAMPAIGN_SCOPES = ["tum", "kategori", "urun"];

function getCampaignsSheet_(create) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(CAMPAIGNS_SHEET_NAME);
  if (!sheet && create) {
    sheet = ss.insertSheet(CAMPAIGNS_SHEET_NAME);
    sheet.getRange(1, 1, 1, CAMPAIGN_COLUMNS.length).setValues([CAMPAIGN_COLUMNS]).setFontWeight("bold");
    sheet.getRange(2, 8, 998, 2).setNumberFormat("@"); // baslangic, bitis metin kalsın
    sheet.setFrozenRows(1);
  }
  return sheet;
}

function todayTR_() { return Utilities.formatDate(new Date(), "Europe/Istanbul", "yyyy-MM-dd"); }

function campDate_(v) {
  if (v instanceof Date) return Utilities.formatDate(v, "Europe/Istanbul", "yyyy-MM-dd");
  var s = String(v == null ? "" : v).trim();
  if (!s) return "";
  var m = s.match(/^(\d{4})-(\d{1,2})-(\d{1,2})/);
  if (m) return m[1] + "-" + ("0" + m[2]).slice(-2) + "-" + ("0" + m[3]).slice(-2);
  m = s.match(/^(\d{1,2})[.\/](\d{1,2})[.\/](\d{4})/);
  if (m) return m[3] + "-" + ("0" + m[2]).slice(-2) + "-" + ("0" + m[1]).slice(-2);
  return "";
}

function getAllCampaigns() {
  var sheet = getCampaignsSheet_(false);
  if (!sheet || sheet.getLastRow() < 2) return [];
  var data = sheet.getDataRange().getValues();
  var headers = data[0];
  return data.slice(1)
    .filter(function (r) { return r[0] && r[2]; })
    .map(function (r) {
      var o = {};
      headers.forEach(function (h, i) { o[h] = r[i]; });
      o.baslangic = campDate_(o.baslangic);
      o.bitis = campDate_(o.bitis);
      o.aktif = /^(1|true|evet)$/i.test(String(o.aktif).trim()) ? "1" : "0";
      return o;
    });
}

function normCampaign_(r) {
  return {
    id: r.id, ad: r.ad || "Kampanya",
    tip: String(r.tip || "").trim().toLowerCase(),
    kapsam: String(r.kapsam || "tum").trim().toLowerCase(),
    hedef: splitList_(r.hedef).map(function (x) { return x.toLowerCase(); }),
    adet: parseInt(r.adet, 10) || 0, deger: parsePrice_(r.deger),
    baslangic: campDate_(r.baslangic), bitis: campDate_(r.bitis),
    aktif: /^(1|true|evet)$/i.test(String(r.aktif == null ? "" : r.aktif).trim())
  };
}

function campActive_(c, today) {
  return c.aktif && (!c.baslangic || c.baslangic <= today) && (!c.bitis || c.bitis >= today);
}

function campTargets_(c, p) {
  if (c.kapsam === "tum") return true;
  if (c.kapsam === "kategori") return c.hedef.indexOf(String(p.category || "").toLowerCase()) > -1;
  if (c.kapsam === "urun") return c.hedef.indexOf(String(p.id || "").toLowerCase()) > -1;
  return false;
}

// Ürün fiyatına uygulanan en düşük fiyatlı kampanya (yoksa null). p: {id, category, price}
function priceCampaign_(p, rawCamps) {
  var today = todayTR_(), best = null;
  (rawCamps || []).forEach(function (raw) {
    var c = normCampaign_(raw);
    if (!campActive_(c, today) || (c.tip !== "yuzde" && c.tip !== "tl") || !(c.deger > 0) || !campTargets_(c, p)) return;
    var np = c.tip === "yuzde" ? Math.round(p.price * (1 - c.deger / 100)) : Math.round(p.price - c.deger);
    np = Math.max(1, np);
    if (np < p.price && (!best || np < best.price)) best = { price: np, camp: c };
  });
  return best;
}

function upsertCampaign(c) {
  if (!c) return { result: "error", message: "Kampanya verisi eksik." };
  var tip = String(c.tip || "").trim().toLowerCase();
  var kapsam = String(c.kapsam || "").trim().toLowerCase();
  if (CAMPAIGN_TYPES.indexOf(tip) < 0) return { result: "error", message: "Geçersiz kampanya türü." };
  if (CAMPAIGN_SCOPES.indexOf(kapsam) < 0) return { result: "error", message: "Geçersiz kapsam." };
  if (!String(c.ad || "").trim()) return { result: "error", message: "Kampanya adı zorunlu." };
  var deger = parsePrice_(c.deger);
  if (!(deger > 0)) return { result: "error", message: "İndirim değeri 0'dan büyük olmalı." };
  if ((tip === "yuzde" || tip === "coklu_yuzde") && deger > 90) return { result: "error", message: "Yüzde en fazla 90 olabilir." };
  var adet = parseInt(c.adet, 10) || 0;
  if ((tip === "coklu_sabit" || tip === "coklu_yuzde") && adet < 2) return { result: "error", message: "Çoklu alım kampanyalarında adet en az 2 olmalı." };
  if (kapsam !== "tum" && !String(c.hedef || "").trim()) return { result: "error", message: "Kategori veya ürün seçmelisiniz." };
  var bas = campDate_(c.baslangic), bit = campDate_(c.bitis);
  if (bas && bit && bit < bas) return { result: "error", message: "Bitiş tarihi başlangıçtan önce olamaz." };

  var lock = LockService.getScriptLock();
  if (!lock.tryLock(15000)) return { result: "error", message: "Başka bir işlem sürüyor, tekrar deneyin." };
  try {
    var sheet = getCampaignsSheet_(true);
    var id = String(c.id || "").trim() || ("k" + new Date().getTime());
    var row = [id, String(c.ad).trim(), tip, kapsam, String(c.hedef || "").trim(), adet || "", deger,
      bas, bit, String(c.aktif) === "0" || c.aktif === false ? "0" : "1", String(c.aciklama || "").trim()];
    var data = sheet.getDataRange().getValues();
    var found = -1;
    for (var i = 1; i < data.length; i++) { if (String(data[i][0]) === id) { found = i + 1; break; } }
    var target = found > -1 ? found : Math.max(sheet.getLastRow() + 1, 2);
    sheet.getRange(target, 8, 1, 2).setNumberFormat("@");
    sheet.getRange(target, 1, 1, row.length).setValues([row]);
    return { result: "ok", id: id };
  } finally { lock.releaseLock(); }
}

function deleteCampaign(id) {
  if (!id) return { result: "error", message: "Kampanya ID zorunlu." };
  var sheet = getCampaignsSheet_(false);
  if (!sheet) return { result: "ok" };
  var data = sheet.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][0]) === String(id)) { sheet.deleteRow(i + 1); break; }
  }
  return { result: "ok" };
}

// Erişim anahtarı ve reklam hesabı numarası Script Properties'te saklanır
// (META_ACCESS_TOKEN, META_AD_ACCOUNT_ID). Panelin "Reklamlar" sekmesinden
// kaydedilir; anahtar hiçbir zaman panele veya siteye geri gönderilmez.
// İLK KURULUMDA: aşağıdaki metaYetkiVer fonksiyonunu bir kez elle çalıştırın.

var META_API = "https://graph.facebook.com/v23.0/";
var META_TOKEN_PROP = "META_ACCESS_TOKEN";
var META_ACCOUNT_PROP = "META_AD_ACCOUNT_ID";
var META_LOG_SHEET = "ReklamGecmisi";
var META_PRESETS = ["today", "yesterday", "last_7d", "last_30d", "this_month", "last_month", "maximum"];
var META_MAX_DAILY_BUDGET = 50000; // TL — yanlışlıkla fazladan sıfır yazılmasına karşı üst sınır
var META_INSIGHT_FIELDS = "spend,impressions,reach,clicks,inline_link_clicks,ctr,cpc,actions,action_values";

// Editörde bir kez çalıştırın: Google'ın "dış bağlantı" iznini onaylamanızı ister.
function metaYetkiVer() {
  UrlFetchApp.fetch("https://graph.facebook.com/", { muteHttpExceptions: true });
  Logger.log("İzin verildi. Şimdi Dağıt > Dağıtımları yönet > yeni sürüm yayınlayın.");
}

function handleMetaAction_(body) {
  try {
    switch (body.action) {
      case "metaStatus": return metaStatus_();
      case "metaSaveConfig": return metaSaveConfig_(body.accountId, body.token);
      case "metaDisconnect": return metaDisconnect_();
      case "metaDashboard": return metaDashboard_(body.preset, body.refresh);
      case "metaSetStatus": return metaSetStatus_(body.id, body.status);
      case "metaSetBudget": return metaSetBudget_(body.id, body.dailyBudget);
      case "metaAdSetup": return metaAdSetup_();
      case "metaSaveAdSetup": return metaSaveAdSetup_(body.pageId, body.pixelId);
      case "metaIgMedia": return metaIgMedia_(body.after);
      case "metaCreateAd": return metaCreateAd_(body.spec || {});
      case "metaCatalogs": return metaCatalogs_();
    }
    return { result: "error", message: "Bilinmeyen işlem." };
  } catch (err) {
    return { result: "error", message: String(err && err.message ? err.message : err) };
  }
}

function metaConfig_() {
  var p = PropertiesService.getScriptProperties();
  var token = p.getProperty(META_TOKEN_PROP), acc = p.getProperty(META_ACCOUNT_PROP);
  return token && acc ? { token: token, account: acc } : null;
}

function metaApi_(path, params, method, tokenOverride) {
  var token = tokenOverride || (metaConfig_() || {}).token;
  if (!token) throw new Error("Meta bağlantısı henüz kurulmadı.");
  params = params || {};
  method = method || "get";
  var url = META_API + path;
  var opts = { method: method, muteHttpExceptions: true, headers: { Authorization: "Bearer " + token } };
  if (method === "get" || method === "delete") {
    var q = Object.keys(params).map(function (k) { return encodeURIComponent(k) + "=" + encodeURIComponent(params[k]); }).join("&");
    if (q) url += "?" + q;
  } else {
    opts.payload = params;
  }
  var res = UrlFetchApp.fetch(url, opts);
  var json;
  try { json = JSON.parse(res.getContentText() || "{}"); }
  catch (e) { throw new Error("Meta'dan beklenmeyen yanıt geldi (" + res.getResponseCode() + ")."); }
  if (json.error) throw new Error(metaErrorText_(json.error));
  return json;
}

function metaErrorText_(e) {
  var code = Number(e.code);
  if (code === 190) return "Meta erişim anahtarı geçersiz veya süresi dolmuş. Reklamlar sekmesinden yeni anahtar girin.";
  if (code === 10 || code === 200 || code === 294 || (code >= 200 && code < 300))
    return "Bu işlem için izin yok. Sistem kullanıcısına reklam hesabında 'Kampanyaları yönet' yetkisi ve ads_read / ads_management izinleri verilmeli.";
  if (code === 4 || code === 17 || code === 32 || code === 613 || code === 80004)
    return "Meta istek sınırına ulaşıldı, birkaç dakika sonra tekrar deneyin.";
  if (code === 100 && /act_/.test(e.message || "")) return "Reklam hesabı bulunamadı. Hesap numarasını kontrol edin.";
  return "Meta: " + (e.error_user_msg || e.message || "bilinmeyen hata");
}

function metaStatus_() {
  var cfg = metaConfig_();
  if (!cfg) return { result: "ok", connected: false };
  var acc = metaApi_("act_" + cfg.account, { fields: "name,currency,account_status" });
  return { result: "ok", connected: true, accountId: cfg.account, name: acc.name, currency: acc.currency, accountStatus: acc.account_status };
}

function metaSaveConfig_(accountId, token) {
  accountId = String(accountId || "").replace(/^act_/i, "").replace(/\D/g, "");
  token = String(token || "").trim();
  if (!accountId) return { result: "error", message: "Reklam hesabı numarası gerekli (sadece rakamlar)." };
  if (token.length < 50) return { result: "error", message: "Erişim anahtarı eksik veya hatalı görünüyor." };
  // Kaydetmeden önce anahtarın gerçekten çalıştığını doğrula
  var acc = metaApi_("act_" + accountId, { fields: "name,currency,account_status" }, "get", token);
  PropertiesService.getScriptProperties().setProperties(
    (function () { var o = {}; o[META_TOKEN_PROP] = token; o[META_ACCOUNT_PROP] = accountId; return o; })()
  );
  metaClearCache_();
  metaLog_("Bağlantı kuruldu", acc.name, "", "");
  return { result: "ok", name: acc.name, currency: acc.currency };
}

function metaDisconnect_() {
  var p = PropertiesService.getScriptProperties();
  p.deleteProperty(META_TOKEN_PROP);
  p.deleteProperty(META_ACCOUNT_PROP);
  p.deleteProperty(META_PAGE_PROP);
  p.deleteProperty(META_IG_PROP);
  p.deleteProperty(META_PIXEL_PROP);
  metaClearCache_();
  metaLog_("Bağlantı kaldırıldı", "", "", "");
  return { result: "ok" };
}

function metaDashboard_(preset, refresh) {
  var cfg = metaConfig_();
  if (!cfg) return { result: "ok", connected: false };
  if (META_PRESETS.indexOf(preset) < 0) preset = "last_7d";
  var cache = CacheService.getScriptCache();
  var key = "metaDash_" + preset;
  if (!refresh) {
    var hit = cache.get(key);
    if (hit) return JSON.parse(hit);
  }
  var act = "act_" + cfg.account;
  var acc = metaApi_(act, { fields: "name,currency,account_status" });
  var sum = metaApi_(act + "/insights", { date_preset: preset, fields: META_INSIGHT_FIELDS });
  var daily = metaApi_(act + "/insights", {
    date_preset: preset, fields: "spend,actions",
    time_increment: preset === "maximum" ? "monthly" : 1, limit: 400
  });
  var camps = metaApi_(act + "/campaigns", {
    fields: "id,name,status,effective_status,objective,daily_budget,lifetime_budget,insights.date_preset(" + preset + "){" + META_INSIGHT_FIELDS + "}",
    limit: 200
  });
  var adsets = metaApi_(act + "/adsets", {
    fields: "id,name,campaign_id,status,effective_status,daily_budget,lifetime_budget,insights.date_preset(" + preset + "){" + META_INSIGHT_FIELDS + "}",
    limit: 400
  });

  var adsetsByCamp = {};
  (adsets.data || []).forEach(function (a) {
    (adsetsByCamp[a.campaign_id] = adsetsByCamp[a.campaign_id] || []).push({
      id: a.id, name: a.name, status: a.status, effectiveStatus: a.effective_status,
      dailyBudget: a.daily_budget ? Number(a.daily_budget) / 100 : null,
      lifetimeBudget: a.lifetime_budget && Number(a.lifetime_budget) > 0 ? Number(a.lifetime_budget) / 100 : null,
      m: metaMetrics_(a.insights && a.insights.data && a.insights.data[0])
    });
  });

  var campaigns = (camps.data || []).map(function (c) {
    var campBudget = !!(c.daily_budget || (c.lifetime_budget && Number(c.lifetime_budget) > 0));
    return {
      id: c.id, name: c.name, status: c.status, effectiveStatus: c.effective_status, objective: c.objective,
      dailyBudget: c.daily_budget ? Number(c.daily_budget) / 100 : null,
      lifetimeBudget: c.lifetime_budget && Number(c.lifetime_budget) > 0 ? Number(c.lifetime_budget) / 100 : null,
      m: metaMetrics_(c.insights && c.insights.data && c.insights.data[0]),
      // Bütçe kampanyada değilse reklam setlerinde yönetilir
      adsets: campBudget ? [] : (adsetsByCamp[c.id] || [])
    };
  });
  campaigns.sort(function (a, b) {
    var aa = a.effectiveStatus === "ACTIVE" ? 0 : 1, bb = b.effectiveStatus === "ACTIVE" ? 0 : 1;
    return aa - bb || b.m.spend - a.m.spend;
  });

  var out = {
    result: "ok", connected: true, preset: preset, updatedAt: new Date().toISOString(),
    account: { id: cfg.account, name: acc.name, currency: acc.currency, status: acc.account_status },
    summary: metaMetrics_(sum.data && sum.data[0]),
    daily: (daily.data || []).map(function (d) {
      return { date: d.date_start, spend: Number(d.spend || 0), purchases: metaAction_(d.actions, META_PURCHASE_TYPES) };
    }),
    campaigns: campaigns
  };
  try { cache.put(key, JSON.stringify(out), 300); } catch (e) { /* büyük yanıt önbelleğe sığmazsa sorun değil */ }
  return out;
}

var META_PURCHASE_TYPES = ["omni_purchase", "purchase", "offsite_conversion.fb_pixel_purchase"];
var META_ATC_TYPES = ["omni_add_to_cart", "add_to_cart", "offsite_conversion.fb_pixel_add_to_cart"];
var META_MSG_TYPES = ["onsite_conversion.messaging_conversation_started_7d"];

function metaAction_(list, types) {
  if (!list || !list.length) return 0;
  for (var t = 0; t < types.length; t++) {
    for (var i = 0; i < list.length; i++) {
      if (list[i].action_type === types[t]) return Number(list[i].value || 0);
    }
  }
  return 0;
}

function metaMetrics_(ins) {
  ins = ins || {};
  var spend = Number(ins.spend || 0);
  var purchaseValue = metaAction_(ins.action_values, META_PURCHASE_TYPES);
  return {
    spend: spend,
    impressions: Number(ins.impressions || 0),
    reach: Number(ins.reach || 0),
    linkClicks: Number(ins.inline_link_clicks || 0),
    ctr: Number(ins.ctr || 0),
    cpc: Number(ins.cpc || 0),
    addToCart: metaAction_(ins.actions, META_ATC_TYPES),
    purchases: metaAction_(ins.actions, META_PURCHASE_TYPES),
    purchaseValue: purchaseValue,
    messages: metaAction_(ins.actions, META_MSG_TYPES),
    roas: spend > 0 ? purchaseValue / spend : 0
  };
}

// Düzenlenecek kampanya/reklam setinin bu reklam hesabına ait olduğunu doğrular
function metaOwnedObject_(id, fields) {
  var cfg = metaConfig_();
  if (!cfg) throw new Error("Meta bağlantısı henüz kurulmadı.");
  if (!/^\d+$/.test(String(id || ""))) throw new Error("Geçersiz kampanya numarası.");
  var obj = metaApi_(String(id), { fields: "account_id,name," + fields });
  if (String(obj.account_id) !== String(cfg.account)) throw new Error("Bu kampanya bağlı reklam hesabına ait değil.");
  return obj;
}

function metaSetStatus_(id, status) {
  if (status !== "ACTIVE" && status !== "PAUSED") return { result: "error", message: "Geçersiz durum." };
  var obj = metaOwnedObject_(id, "status");
  metaApi_(String(id), { status: status }, "post");
  metaClearCache_();
  metaLog_(status === "ACTIVE" ? "Başlatıldı" : "Durduruldu", obj.name, obj.status, status);
  return { result: "ok" };
}

function metaSetBudget_(id, dailyBudget) {
  var tl = Number(dailyBudget);
  if (!(tl > 0)) return { result: "error", message: "Geçerli bir günlük bütçe girin." };
  if (tl > META_MAX_DAILY_BUDGET) return { result: "error", message: "Günlük bütçe " + META_MAX_DAILY_BUDGET + " TL'yi geçemez (güvenlik sınırı)." };
  var obj = metaOwnedObject_(id, "daily_budget");
  if (!obj.daily_budget) return { result: "error", message: "Bu kampanyada günlük bütçe yok (toplam bütçe kullanılıyor). Reklam Yöneticisi'nden düzenleyin." };
  var oldTl = Number(obj.daily_budget) / 100;
  metaApi_(String(id), { daily_budget: String(Math.round(tl * 100)) }, "post");
  metaClearCache_();
  metaLog_("Günlük bütçe değişti", obj.name, oldTl + " TL", tl + " TL");
  return { result: "ok", oldBudget: oldTl, newBudget: tl };
}

function metaClearCache_() {
  CacheService.getScriptCache().removeAll(META_PRESETS.map(function (p) { return "metaDash_" + p; }));
}

// Panelden yapılan her değişiklik "ReklamGecmisi" sekmesine not edilir
function metaLog_(action, name, before, after) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sh = ss.getSheetByName(META_LOG_SHEET);
    if (!sh) { sh = ss.insertSheet(META_LOG_SHEET); sh.appendRow(["Tarih", "İşlem", "Kampanya / Reklam Seti", "Önce", "Sonra"]); }
    sh.appendRow([new Date(), action, name || "", before || "", after || ""]);
  } catch (e) { console.error("Reklam geçmişi yazılamadı: " + e); }
}

var BLOG_SHEET_NAME = "Blog";
var BLOG_COLUMNS = ["slug", "title", "metaTitle", "metaDescription", "focusKeyword", "excerpt", "coverImage",
  "content", "relatedProducts", "tags", "date", "updated", "published"];

function getBlogSheet_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(BLOG_SHEET_NAME);
  if (!sh) {
    sh = ss.insertSheet(BLOG_SHEET_NAME);
    sh.getRange(1, 1, 1, BLOG_COLUMNS.length).setValues([BLOG_COLUMNS]);
    sh.setFrozenRows(1);
  }
  return sh;
}

function getAllBlog_() {
  var data = getBlogSheet_().getDataRange().getDisplayValues();
  var headers = data[0];
  return data.slice(1).filter(function (r) { return r[0]; }).map(function (r) {
    var o = {};
    headers.forEach(function (h, i) { o[h] = r[i]; });
    return o;
  });
}

function blogToday_() {
  return Utilities.formatDate(new Date(), "Europe/Istanbul", "yyyy-MM-dd");
}

function upsertBlog_(post) {
  if (!post) return { result: "error", message: "Yazı verisi eksik." };
  var slug = String(post.slug || "").trim().toLowerCase();
  if (!/^[a-z0-9]+(-[a-z0-9]+)*$/.test(slug)) return { result: "error", message: "Adres (slug) sadece küçük harf, rakam ve tire içerebilir." };
  if (!String(post.title || "").trim()) return { result: "error", message: "Başlık zorunlu." };
  if (String(post.content || "").length > 45000) return { result: "error", message: "Yazı çok uzun (tablo hücresi sınırı). Lütfen iki yazıya bölün." };

  var sh = getBlogSheet_();
  var data = sh.getDataRange().getDisplayValues();
  var headers = data[0];
  BLOG_COLUMNS.forEach(function (c) {
    if (headers.indexOf(c) < 0) { headers.push(c); sh.getRange(1, headers.length).setValue(c); }
  });
  var slugCol = headers.indexOf("slug");
  var row = -1, origSlug = String(post.originalSlug || slug);
  for (var i = 1; i < data.length; i++) {
    var s = String(data[i][slugCol]);
    if (s === origSlug) row = i + 1;
    else if (s === slug) return { result: "error", message: "Bu adresle başka bir yazı var." };
  }
  var old = row > -1 ? data[row - 1] : [];
  var today = blogToday_();
  post.slug = slug;
  post.updated = today;
  if (!post.date) post.date = row > -1 ? (old[headers.indexOf("date")] || today) : today;
  post.published = /^(1|true)$/i.test(String(post.published)) ? "1" : "0";

  var values = headers.map(function (h, c) {
    if (post[h] !== undefined) return String(post[h]);
    return row > -1 ? (old[c] == null ? "" : old[c]) : "";
  });
  var target = row > -1 ? sh.getRange(row, 1, 1, values.length) : sh.getRange(sh.getLastRow() + 1, 1, 1, values.length);
  target.setNumberFormat("@").setValues([values]);
  return { result: "ok", slug: slug };
}

function deleteBlog_(slug) {
  if (!slug) return { result: "error", message: "Adres eksik." };
  var sh = getBlogSheet_();
  var data = sh.getDataRange().getDisplayValues();
  var slugCol = data[0].indexOf("slug");
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][slugCol]) === String(slug)) { sh.deleteRow(i + 1); return { result: "ok" }; }
  }
  return { result: "error", message: "Yazı bulunamadı." };
}

var META_PAGE_PROP = "META_PAGE_ID";
var META_IG_PROP = "META_IG_USER_ID";
var META_PIXEL_PROP = "META_PIXEL_ID";
var META_MAX_AD_DAYS = 90;

function metaAdSetup_() {
  var cfg = metaConfig_();
  if (!cfg) return { result: "ok", connected: false };
  var props = PropertiesService.getScriptProperties();
  var pages = (metaApi_("me/accounts", { fields: "id,name,instagram_business_account{id,username}", limit: 50 }).data || [])
    .map(function (p) {
      var ig = p.instagram_business_account || {};
      return { id: p.id, name: p.name, igId: ig.id || "", igUsername: ig.username || "" };
    });
  var pixels = [];
  try { pixels = (metaApi_("act_" + cfg.account + "/adspixels", { fields: "id,name", limit: 25 }).data || []).map(function (x) { return { id: x.id, name: x.name }; }); }
  catch (e) { /* piksel listesi okunamazsa ayardaki piksel kullanılır */ }

  var pageId = props.getProperty(META_PAGE_PROP) || "";
  var page = pages.filter(function (p) { return p.id === pageId; })[0];
  if (!page && pages.length) {
    page = pages.filter(function (p) { return p.igId; })[0] || pages[0];
    props.setProperty(META_PAGE_PROP, page.id);
    if (page.igId) props.setProperty(META_IG_PROP, page.igId); else props.deleteProperty(META_IG_PROP);
  }
  var pixelId = props.getProperty(META_PIXEL_PROP) || String(getAllSettings().metaPixelId || "").replace(/\D/g, "") || (pixels[0] ? pixels[0].id : "");
  return { result: "ok", connected: true, pages: pages, pageId: page ? page.id : "", igId: page ? page.igId : "", igUsername: page ? page.igUsername : "", pixels: pixels, pixelId: pixelId };
}

function metaSaveAdSetup_(pageId, pixelId) {
  var pages = metaApi_("me/accounts", { fields: "id,instagram_business_account{id}", limit: 50 }).data || [];
  var page = pages.filter(function (p) { return p.id === String(pageId); })[0];
  if (!page) return { result: "error", message: "Bu Facebook sayfası sistem kullanıcısına atanmamış." };
  var props = PropertiesService.getScriptProperties();
  props.setProperty(META_PAGE_PROP, page.id);
  if (page.instagram_business_account) props.setProperty(META_IG_PROP, page.instagram_business_account.id); else props.deleteProperty(META_IG_PROP);
  if (pixelId) props.setProperty(META_PIXEL_PROP, String(pixelId).replace(/\D/g, ""));
  return { result: "ok" };
}

function metaIgMedia_(after) {
  var ig = PropertiesService.getScriptProperties().getProperty(META_IG_PROP);
  if (!ig) return { result: "error", message: "Seçili Facebook sayfasına bağlı bir Instagram işletme hesabı bulunamadı." };
  var params = { fields: "id,caption,media_type,media_product_type,thumbnail_url,media_url,permalink,timestamp,like_count,comments_count", limit: 24 };
  if (after) params.after = after;
  var res = metaApi_(ig + "/media", params);
  return {
    result: "ok",
    items: (res.data || []).map(function (m) {
      return { id: m.id, caption: m.caption || "", type: m.media_product_type === "REELS" ? "REELS" : m.media_type,
        thumb: m.thumbnail_url || m.media_url || "", permalink: m.permalink, date: m.timestamp, likes: m.like_count || 0, comments: m.comments_count || 0 };
    }),
    next: res.paging && res.paging.next && res.paging.cursors ? res.paging.cursors.after : ""
  };
}

// Ürün fotoğrafını Meta'ya yükler, görsel kodunu (hash) döndürür
function metaUploadImage_(act, url) {
  var r = UrlFetchApp.fetch(url, { muteHttpExceptions: true });
  if (r.getResponseCode() !== 200) throw new Error("Görsel indirilemedi: " + url);
  var res = metaApi_(act + "/adimages", { bytes: Utilities.base64Encode(r.getBlob().getBytes()) }, "post");
  var keys = Object.keys(res.images || {});
  if (!keys.length || !res.images[keys[0]].hash) throw new Error("Görsel Meta'ya yüklenemedi.");
  return res.images[keys[0]].hash;
}

function metaCreateAd_(spec) {
  var cfg = metaConfig_();
  if (!cfg) return { result: "error", message: "Meta bağlantısı henüz kurulmadı." };
  var props = PropertiesService.getScriptProperties();
  var pageId = props.getProperty(META_PAGE_PROP), igId = props.getProperty(META_IG_PROP) || "";
  if (!pageId) return { result: "error", message: "Önce reklamın yayınlanacağı Facebook sayfasını seçin." };
  var act = "act_" + cfg.account;
  var kind = ["igpost", "igmulti", "catalog"].indexOf(spec.kind) > -1 ? spec.kind : "product";
  var goal = kind === "catalog" ? "catalog" : String(spec.goal || "traffic");
  var allowed = { igpost: ["reach", "traffic", "igdm"], product: ["sales", "traffic", "whatsapp"], igmulti: ["sales", "traffic", "whatsapp"], catalog: ["catalog"] }[kind];
  if (allowed.indexOf(goal) < 0) return { result: "error", message: "Bu reklam türü için geçersiz hedef." };

  var budget = Number(spec.dailyBudget);
  if (!(budget >= 1)) return { result: "error", message: "Geçerli bir günlük bütçe girin." };
  if (budget > META_MAX_DAILY_BUDGET) return { result: "error", message: "Günlük bütçe " + META_MAX_DAILY_BUDGET + " TL'yi geçemez (güvenlik sınırı)." };
  var days = Math.max(0, Math.min(META_MAX_AD_DAYS, parseInt(spec.days, 10) || 0));
  var ageMin = Math.max(18, Math.min(65, parseInt(spec.ageMin, 10) || 18));
  var ageMax = Math.max(ageMin, Math.min(65, parseInt(spec.ageMax, 10) || 65));
  var link = String(spec.link || SITE_URL + "/").trim();
  if (!/^https:\/\//i.test(link)) return { result: "error", message: "Bağlantı https:// ile başlamalı." };
  var pixelId = props.getProperty(META_PIXEL_PROP) || String(getAllSettings().metaPixelId || "").replace(/\D/g, "");
  if (kind === "catalog" && !(/^\d+$/.test(String(spec.catalogId || "")) && /^\d+$/.test(String(spec.productSetId || ""))))
    return { result: "error", message: "Katalog ve ürün grubu seçin." };
  if (goal === "sales" && !pixelId) return { result: "error", message: "Satış hedefi için Meta Pixel gerekli. Ayarlar sekmesinden Pixel ID girin ya da 'Siteye trafik' hedefini seçin." };
  if (goal === "igdm" && !igId) return { result: "error", message: "Instagram mesaj hedefi için sayfaya bağlı Instagram hesabı gerekli." };

  var stamp = Utilities.formatDate(new Date(), "Europe/Istanbul", "yyyy-MM-dd HH:mm");
  var baseName = "Panel | " + String(spec.name || (kind === "igpost" ? "Instagram gönderisi" : "Ürün")).slice(0, 60) + " | " + stamp;
  var created = {};

  try {
    // 1) Reklam görseli/metni (önce bu: hata verirse hiçbir şey oluşmaz)
    var creativeParams = { name: baseName };
    if (kind === "igpost") {
      if (!/^\d+$/.test(String(spec.mediaId || ""))) throw new Error("Instagram gönderisi seçilmedi.");
      if (!igId) throw new Error("Seçili sayfaya bağlı Instagram hesabı yok.");
      creativeParams.object_id = pageId;
      creativeParams.instagram_user_id = igId;
      creativeParams.source_instagram_media_id = String(spec.mediaId);
      if (goal === "traffic") creativeParams.call_to_action = JSON.stringify({ type: "LEARN_MORE", value: { link: link } });
      if (goal === "igdm") creativeParams.call_to_action = JSON.stringify({ type: "MESSAGE_PAGE", value: { app_destination: "INSTAGRAM_DIRECT" } });
    } else if (kind === "catalog") {
      var tpl = {
        message: String(spec.primaryText || "").slice(0, 2000),
        name: "{{product.name}}",
        description: "{{product.current_price strip_zeros}} · Kapıda ödeme",
        link: SITE_URL + "/",
        call_to_action: { type: "SHOP_NOW" },
        multi_share_end_card: false
      };
      var cstory = { page_id: pageId, template_data: tpl };
      if (igId) cstory.instagram_user_id = igId;
      creativeParams.object_story_spec = JSON.stringify(cstory);
      creativeParams.product_set_id = String(spec.productSetId);
    } else if (kind === "igmulti") {
      var ids = (spec.mediaIds || []).filter(function (x) { return /^\d+$/.test(String(x)); }).slice(0, 10);
      if (ids.length < 2) throw new Error("Kaydırmalı reklam için en az 2 gönderi seçin.");
      var mLink = goal === "whatsapp" ? "https://api.whatsapp.com/send" : link;
      var mCta = goal === "whatsapp" ? { type: "WHATSAPP_MESSAGE", value: { app_destination: "WHATSAPP" } } : { type: "SHOP_NOW", value: { link: link } };
      var mHead = String(spec.headline || "").slice(0, 120);
      var cards = ids.map(function (id) { return metaIgMediaToCard_(act, id, mLink, mHead, mCta); });
      var mStory = { page_id: pageId, link_data: { message: String(spec.primaryText || "").slice(0, 2000), link: mLink, call_to_action: mCta,
        child_attachments: cards, multi_share_optimized: true, multi_share_end_card: false } };
      if (igId) mStory.instagram_user_id = igId;
      creativeParams.object_story_spec = JSON.stringify(mStory);
    } else {
      var images = (spec.images || []).filter(function (u) { return /^https:\/\//i.test(u) && !/\.(mp4|mov|webm)(\?|$)/i.test(u); }).slice(0, 10);
      if (!images.length) throw new Error("En az bir ürün fotoğrafı seçin.");
      var message = String(spec.primaryText || "").slice(0, 2000);
      var headline = String(spec.headline || spec.name || "").slice(0, 120);
      var descr = String(spec.description || "").slice(0, 120);
      var waLink = "https://api.whatsapp.com/send";
      var cta = goal === "whatsapp"
        ? { type: "WHATSAPP_MESSAGE", value: { app_destination: "WHATSAPP" } }
        : { type: "SHOP_NOW", value: { link: link } };
      var hashes = images.map(function (u) { return metaUploadImage_(act, u); });
      var linkData = { message: message, link: goal === "whatsapp" ? waLink : link, call_to_action: cta };
      if (hashes.length === 1) {
        linkData.image_hash = hashes[0]; linkData.name = headline; if (descr) linkData.description = descr;
      } else {
        linkData.child_attachments = hashes.map(function (h) {
          var c = { link: goal === "whatsapp" ? waLink : link, image_hash: h, name: headline, call_to_action: cta };
          if (descr) c.description = descr;
          return c;
        });
        linkData.multi_share_optimized = true;
        linkData.multi_share_end_card = false;
      }
      var story = { page_id: pageId, link_data: linkData };
      if (igId) story.instagram_user_id = igId;
      creativeParams.object_story_spec = JSON.stringify(story);
    }
    var creative = metaApi_(act + "/adcreatives", creativeParams, "post");
    created.creativeId = creative.id;

    // 2) Kampanya — kampanya bütçesiyle, DURAKLATILMIŞ
    var objective = { sales: "OUTCOME_SALES", catalog: "OUTCOME_SALES", traffic: "OUTCOME_TRAFFIC", whatsapp: "OUTCOME_ENGAGEMENT", igdm: "OUTCOME_ENGAGEMENT", reach: "OUTCOME_AWARENESS" }[goal];
    var campParams = {
      name: baseName, objective: objective, status: "PAUSED", buying_type: "AUCTION",
      special_ad_categories: "[]", daily_budget: String(Math.round(budget * 100)), bid_strategy: "LOWEST_COST_WITHOUT_CAP"
    };
    if (kind === "catalog") campParams.promoted_object = JSON.stringify({ product_catalog_id: String(spec.catalogId) });
    var camp = metaApi_(act + "/campaigns", campParams, "post");
    created.campaignId = camp.id;

    // 3) Reklam seti — kitle ve hedef
    var targeting = { geo_locations: { countries: ["TR"] }, age_min: ageMin, age_max: ageMax, targeting_automation: { advantage_audience: 0 } };
    if (spec.gender !== "all") targeting.genders = [2];
    var adset = { name: baseName, campaign_id: camp.id, status: "ACTIVE", billing_event: "IMPRESSIONS", targeting: JSON.stringify(targeting) };
    if (goal === "sales") { adset.optimization_goal = "OFFSITE_CONVERSIONS"; adset.promoted_object = JSON.stringify({ pixel_id: pixelId, custom_event_type: "PURCHASE" }); }
    if (goal === "traffic") { adset.optimization_goal = "LINK_CLICKS"; adset.destination_type = "WEBSITE"; }
    if (goal === "whatsapp") { adset.optimization_goal = "CONVERSATIONS"; adset.destination_type = "WHATSAPP"; adset.promoted_object = JSON.stringify({ page_id: pageId }); }
    if (goal === "igdm") { adset.optimization_goal = "CONVERSATIONS"; adset.destination_type = "INSTAGRAM_DIRECT"; adset.promoted_object = JSON.stringify({ page_id: pageId }); }
    if (goal === "reach") { adset.optimization_goal = "REACH"; }
    if (goal === "catalog") {
      adset.optimization_goal = "OFFSITE_CONVERSIONS";
      adset.promoted_object = JSON.stringify({ product_set_id: String(spec.productSetId), custom_event_type: "PURCHASE" });
      var purchaseExcl = [{ retention_seconds: 3 * 86400, rule: { event: { eq: "Purchase" } } }];
      var ct;
      if (spec.audience === "retarget") {
        var rs = Math.max(3, Math.min(30, parseInt(spec.retargetDays, 10) || 14)) * 86400;
        ct = { geo_locations: { countries: ["TR"] }, targeting_automation: { advantage_audience: 0 },
          product_audience_specs: [{ product_set_id: String(spec.productSetId),
            inclusions: [{ retention_seconds: rs, rule: { event: { eq: "ViewContent" } } }, { retention_seconds: rs, rule: { event: { eq: "AddToCart" } } }],
            exclusions: purchaseExcl }] };
      } else {
        ct = targeting;
        ct.excluded_product_audience_specs = [{ product_set_id: String(spec.productSetId), inclusions: purchaseExcl }];
      }
      adset.targeting = JSON.stringify(ct);
    }
    if (days > 0) adset.end_time = new Date(Date.now() + days * 86400000).toISOString();
    var as = metaApi_(act + "/adsets", adset, "post");
    created.adsetId = as.id;

    // 4) Reklam
    var ad = metaApi_(act + "/ads", { name: baseName, adset_id: as.id, creative: JSON.stringify({ creative_id: creative.id }), status: "ACTIVE" }, "post");
    created.adId = ad.id;
  } catch (err) {
    // Yarım kalan kampanyayı temizle (kampanya silinince reklam seti ve reklam da silinir)
    if (created.campaignId) { try { metaApi_(created.campaignId, {}, "delete"); } catch (e) {} }
    throw err;
  }

  // Önizlemeler (Instagram akış, Reels ve Facebook akış)
  var previews = [];
  ["INSTAGRAM_STANDARD", "INSTAGRAM_REELS", "MOBILE_FEED_STANDARD"].forEach(function (fmt) {
    try {
      var pv = metaApi_(created.creativeId + "/previews", { ad_format: fmt });
      if (pv.data && pv.data[0] && pv.data[0].body) previews.push({ format: fmt, html: pv.data[0].body });
    } catch (e) { /* bu yerleşim için önizleme yoksa atla */ }
  });

  metaClearCache_();
  metaLog_("Reklam oluşturuldu (duraklatılmış)", baseName, "", budget + " TL/gün" + (days ? ", " + days + " gün" : ""));
  return { result: "ok", campaignId: created.campaignId, name: baseName, previews: previews };
}

// Instagram gönderisinin görselini/videosunu Meta reklam kütüphanesine yükleyip kaydırmalı reklam kartına çevirir
function metaIgMediaToCard_(act, mediaId, link, headline, cta) {
  var m = metaApi_(String(mediaId), { fields: "media_type,media_url,thumbnail_url,children{media_type,media_url,thumbnail_url}" });
  if (m.media_type === "CAROUSEL_ALBUM" && m.children && m.children.data && m.children.data.length) m = m.children.data[0];
  var card = { link: link, call_to_action: cta };
  if (headline) card.name = headline;
  if (m.media_type === "VIDEO") {
    if (!m.media_url) throw new Error("Instagram videosunun dosyasına erişilemedi.");
    var vid = metaApi_(act + "/advideos", { file_url: m.media_url }, "post").id;
    metaWaitVideo_(vid);
    card.video_id = vid;
    card.image_hash = metaUploadImage_(act, m.thumbnail_url);
  } else {
    if (!m.media_url) throw new Error("Instagram görseline erişilemedi.");
    card.image_hash = metaUploadImage_(act, m.media_url);
  }
  return card;
}

// Yüklenen videonun Meta tarafında işlenmesini bekler (en fazla ~2 dakika)
function metaWaitVideo_(videoId) {
  for (var i = 0; i < 24; i++) {
    var st = metaApi_(String(videoId), { fields: "status" }).status || {};
    if (st.video_status === "ready") return;
    if (st.video_status === "error") throw new Error("Video Meta tarafından işlenemedi.");
    Utilities.sleep(5000);
  }
  throw new Error("Video işlenmesi uzun sürdü; birkaç dakika sonra tekrar deneyin.");
}

// İşletmenin ürün katalogları ve ürün grupları
function metaCatalogs_() {
  var cfg = metaConfig_();
  if (!cfg) return { result: "error", message: "Meta bağlantısı henüz kurulmadı." };
  var acc = metaApi_("act_" + cfg.account, { fields: "business{id,name}" });
  if (!acc.business) return { result: "error", message: "Reklam hesabı bir işletme portföyüne bağlı değil." };
  var cats = [];
  ["owned_product_catalogs", "client_product_catalogs"].forEach(function (edge) {
    try {
      (metaApi_(acc.business.id + "/" + edge, { fields: "id,name,product_count", limit: 25 }).data || []).forEach(function (c) {
        if (!cats.some(function (x) { return x.id === c.id; })) cats.push({ id: c.id, name: c.name, count: c.product_count || 0 });
      });
    } catch (e) { /* bu tür katalog yoksa atla */ }
  });
  cats.forEach(function (c) {
    try { c.sets = (metaApi_(c.id + "/product_sets", { fields: "id,name,product_count", limit: 50 }).data || []).map(function (x) { return { id: x.id, name: x.name, count: x.product_count || 0 }; }); }
    catch (e) { c.sets = []; }
  });
  if (!cats.length) return { result: "error", message: "Katalog bulunamadı. Sistem kullanıcısına İşletme Ayarları > Varlık ata > Kataloglar'dan kataloğunu ata." };
  return { result: "ok", catalogs: cats };
}
